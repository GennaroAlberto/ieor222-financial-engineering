% Chapter 18: Hawkes processes and limit order books.

At the scale of milliseconds a market is not a diffusion but a stream of events: limit orders are
posted and cancelled, market orders execute against them, and the price moves by whole ticks. The
events cluster. A trade makes further trades more likely in the next seconds, a cancellation is
often followed by a new order, a burst of activity feeds on itself. \emph{Hawkes processes} are point
processes in which every event raises the rate of future events. They are tractable (a branching
structure, an explicit likelihood, an exact simulation algorithm) and, fitted to high-frequency data,
they often turn out to sit close to the boundary of stability.

This chapter builds the toolkit (intensities, the likelihood, the time-change theorem), the
statistics (maximum likelihood and goodness of fit), and the application to order flow; it ends with
the scaling limits of Jaisson and Rosenbaum. Hawkes processes close to instability look, on a large
time scale, like the CIR process of \cref{prop:ir-cir}, or, when the excitation decays slowly, like
the rough processes of \cref{ch:19-rough-volatility}. A price driven by such order flow becomes a
(rough) Heston model. The survey of \textcite{bacry-mastromatteo-muzy-2015} covers the financial
applications, and \textcite{daley-vere-jones-2003} the theory of point processes.

\paragraph{Setting.} $(\Omega,\F,\Prob)$ is a probability space. Every point process starts at time
$0$ with no events before. For functions on $[0,\infty)$, $(f*g)(t)=\int_0^tf(t-s)g(s)\,\dd s$; for
matrix-valued functions the products inside are matrix products. $\varrho(G)$ is the spectral radius
(largest modulus of an eigenvalue) of a square matrix $G$.

\section{Point processes and intensities}\label{sec:hk-intensity}

A \emph{$d$-variate point process} is a sequence of event times $0<t_1<t_2<\cdots$ with types
$\kappa_k\in\{1,\dots,d\}$; we allow $t_k=\infty$ (then $t_{k+1}=\infty$: there are only $k-1$
events). Its \emph{counting processes} are $N^i_t=\#\{k:t_k\le t,\ \kappa_k=i\}$ and
$N_t=\sum_iN^i_t$, and $H_k=(t_1,\kappa_1,\dots,t_k,\kappa_k)$ is the history of the first $k$ events.
The process \emph{explodes} if $t_\infty=\lim_kt_k<\infty$.

\begin{definition}[Conditional intensity]\label{def:hk-intensity}
For each type $i$, let $\lambda^i(t\mid h)\ge0$ be defined for every finite history
$h=(s_1,j_1,\dots,s_k,j_k)$ with $s_1<\dots<s_k$ and every $t>s_k$ (with $s_0=0$), measurable in
$(t,h)$ and integrable in $t$ on bounded intervals, and let $\bar\lambda=\sum_i\lambda^i$. A point process
has \emph{conditional intensity} $(\lambda^i)_{i\le d}$ if, for every $k\ge0$, on $\{t_k<\infty\}$,
\begin{equation}\label{eq:hk-hazard}
  \Prob\big(t_{k+1}\in\dd t,\ \kappa_{k+1}=i\,\big|\,H_k\big)
  =\lambda^i(t\mid H_k)\exp\Big(-\int_{t_k}^t\bar\lambda(u\mid H_k)\,\dd u\Big)\dd t,\qquad t>t_k .
\end{equation}
We write $\lambda^i(t)=\lambda^i(t\mid H_{k})$ for $t_k<t\le t_{k+1}$: the intensity given the events strictly
before $t$.
\end{definition}

Summing \eqref{eq:hk-hazard} over $i$ and integrating, $\Prob(t_{k+1}>t\mid H_k)=\exp(-\int_{t_k}^t\bar\lambda(u\mid
H_k)\dd u)$: the next event comes at the \emph{hazard rate} $\bar\lambda$, and it is of type $i$ with probability
$\lambda^i/\bar\lambda$ at that time. Such a process always exists: with \iid exponential variables $E_k$ and
uniform $U_k$, set $t_{k+1}=\inf\{t>t_k:\int_{t_k}^t\bar\lambda(u\mid H_k)\dd u\ge E_{k+1}\}$ and pick the type
by $U_{k+1}$ (\textcite{daley-vere-jones-2003}, Algorithm~7.4.III). A Poisson process of rate $\mu$ is the case
$\lambda\equiv\mu$, with \iid $\mathrm{Exp}(\mu)$ gaps; an inhomogeneous one, with rate $h(t)$, the case
$\lambda(t\mid h)=h(t)$.

\begin{remark}[Intensities and martingales]\label{rem:hk-martingale}
Heuristically, $\Prob(\text{a type-}i\text{ event in }(t,t+\delta]\mid\text{past})=\lambda^i(t)\delta+o(\delta)$. In
the general theory the intensity is defined through martingales: $N^i_t-\int_0^t\lambda^i(s)\,\dd s$ is a (local)
martingale for the filtration generated by the process, and this property determines the law of the process
(\textcite{daley-vere-jones-2003}, Lemma~7.2.V and Proposition~7.2.IV). We only use its consequence for
expectations, proved directly below.
\end{remark}

\begin{lemma}[Compensator and density]\label{lem:hk-compensator}
Let $N$ have conditional intensity $(\lambda^i)$.
\begin{enumerate}[label=(\roman*)]
  \item For $t\ge0$, $k\ge1$ and every $i$, $\E[N^i_{t\wedge t_k}]=\E\int_0^{t\wedge t_k}\lambda^i(s)\,\dd s$, and
    $\E[N^i_t]=\E\int_0^{t\wedge t_\infty}\lambda^i(s)\,\dd s$ in $[0,\infty]$.
  \item If $\E[N_t]<\infty$ for every $t$, then $N$ does not explode, and
    $\E\big[\sum_{k:\kappa_k=i}g(t_k)\big]=\int_0^\infty g(s)\,\E[\lambda^i(s)]\,\dd s$ for every Borel $g\ge0$.
  \item For $T>0$, $n\in\N$, $0<s_1<\dots<s_n\le T$ and types $j_1,\dots,j_n$, writing $\lambda^i$ for the
    intensity of the configuration $h=(s_1,j_1,\dots,s_n,j_n)$,
    \begin{equation}\label{eq:hk-density}
      \Prob\big(N_T=n,\ t_k\in\dd s_k,\ \kappa_k=j_k,\ k\le n\big)
      =\prod_{k=1}^n\lambda^{j_k}(s_k)\ \exp\Big(-\int_0^T\bar\lambda(u)\,\dd u\Big)\,\dd s_1\cdots\dd s_n .
    \end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
(i) Fix $k\ge1$ and write $H=H_{k-1}$. On $(t_{k-1},t_k]$ the intensity is $\lambda^i(\cdot\mid H)$, and
$\Prob(t_k>s\mid H)=e^{-\int_{t_{k-1}}^s\bar\lambda(u\mid H)\dd u}$. By \eqref{eq:hk-hazard} and Tonelli, on
$\{t_{k-1}<t\}$,
\[
  \Prob(t_k\le t,\ \kappa_k=i\mid H)=\int_{t_{k-1}}^t\lambda^i(s\mid H)\,\Prob(t_k>s\mid H)\,\dd s
  =\E\Big[\int_{t_{k-1}}^{t_k\wedge t}\lambda^i(s)\,\dd s\,\Big|\,H\Big],
\]
and both sides vanish on $\{t_{k-1}\ge t\}$. Take expectations and sum over the first $k$ events: the
intervals $(t_{l-1},t_l]$ partition $(0,t_k]$. Letting $k\to\infty$ gives the second identity by monotone
convergence, since $N^i_{t\wedge t_k}\uparrow N^i_t$.

(ii) If $\E[N_t]<\infty$, then $N_t<\infty$ almost surely, so $t_\infty>t$; taking $t=1,2,\dots$ gives
$t_\infty=\infty$ almost surely. By (i) and Tonelli, $\E[N^i_b-N^i_a]=\int_a^b\E[\lambda^i(s)]\,\dd s$ for $a<b$.
The measures $A\mapsto\E[N^i(A)]$ and $A\mapsto\int_A\E[\lambda^i]$ agree on intervals and are finite on
bounded sets, so they agree on Borel sets (Dynkin's $\pi$--$\lambda$ theorem); approximate $g$ by simple
functions and use monotone convergence.

(iii) Apply \eqref{eq:hk-hazard} to $k=0,\dots,n-1$ in turn and multiply by $\Prob(t_{n+1}>T\mid
H_n)=e^{-\int_{s_n}^T\bar\lambda}$; the exponentials combine into $e^{-\int_0^T\bar\lambda}$.
\end{proof}

\section{The Hawkes process}\label{sec:hk-hawkes}

\begin{definition}[Hawkes process]\label{def:hk-hawkes}
Let $d\ge1$, $\mu\in[0,\infty)^d$, and let $\varphi_{ij}:(0,\infty)\to[0,\infty)$ be Borel with
$\|\varphi_{ij}\|_1=\int_0^\infty\varphi_{ij}(t)\,\dd t<\infty$. A \emph{$d$-variate Hawkes process} with
\emph{baseline} $\mu$ and \emph{kernels} $\varphi_{ij}$ is a point process with conditional intensity
\begin{equation}\label{eq:hk-hawkes}
  \lambda^i(t)=\mu_i+\sum_{j=1}^d\ \sum_{k:\,\kappa_k=j,\ t_k<t}\varphi_{ij}(t-t_k)
  =\mu_i+\sum_{j=1}^d\int_{(0,t)}\varphi_{ij}(t-s)\,\dd N^j_s .
\end{equation}
Its \emph{branching matrix} is $G=(\|\varphi_{ij}\|_1)_{i,j}$; for $d=1$, $\|\varphi\|_1$ is the \emph{branching
ratio}. \emph{Exponential kernels} are $\varphi_{ij}(t)=\alpha_{ij}\beta_{ij}e^{-\beta_{ij}t}$ with
$\alpha_{ij}\ge0$, $\beta_{ij}>0$, so that $G=(\alpha_{ij})$.
\end{definition}

\textcite{hawkes-1971} introduced these processes on the whole real line and computed their mean rate and
second-order structure. An event of type $j$ raises the intensity of type $i$ by $\varphi_{ij}$; the entry
$G_{ij}$ is the total extra number of type-$i$ events it causes directly (\cref{prop:hk-branching}).

\begin{remark}[Exponential kernels]\label{rem:hk-exp}
For $d=1$ and $\varphi(t)=\alpha\beta e^{-\beta t}$, between events $\lambda(t)=\mu+(\lambda(s)-\mu)e^{-\beta(t-s)}$
relaxes towards $\mu$ at rate $\beta$, and at each event it jumps by $\alpha\beta$. The intensity is therefore a
Markov process, and the past enters through one number. For $d\ge2$ the $d^2$ sums
$R_{ij}(t)=\sum_{\kappa_k=j,\,t_k<t}e^{-\beta_{ij}(t-t_k)}$ play this role. In the stationary regime the
covariance density of the process decays like $e^{-\beta(1-\alpha)\tau}$ (\textcite{hawkes-1971}, Section~3), and
the mean intensity of the process started empty approaches its limit at the same rate
(\exref{ex:hk-exponential}): the memory becomes long as $\alpha\uparrow1$.
\end{remark}

\begin{proposition}[Mean intensity and stability]\label{prop:hk-stationary}
Let $N$ be a Hawkes process, $\Phi=(\varphi_{ij})$, $\Phi^{*1}=\Phi$, $\Phi^{*(n+1)}=\Phi*\Phi^{*n}$, and
\[
  m(t)=\mu+\sum_{n\ge1}\Big(\int_0^t\Phi^{*n}(s)\,\dd s\Big)\mu\in[0,\infty]^d .
\]
\begin{enumerate}[label=(\roman*)]
  \item $N$ does not explode, $\E[N_t]=\int_0^tm(s)\,\dd s<\infty$ and $\E[\lambda(t)]=m(t)$ for every $t$. The
    function $m$ is finite and nondecreasing, and it is the only locally bounded solution of the renewal
    equation $m=\mu+\Phi*m$.
  \item If $\varrho(G)<1$, then $m(t)\uparrow\Lambda=(I-G)^{-1}\mu=\sum_{n\ge0}G^n\mu$ as $t\to\infty$, and
    $\E[N_t]/t\to\Lambda$. The total rate of events tends to $\1^\top(I-G)^{-1}\mu$.
  \item If $\varrho(G)\ge1$ and $\mu_i>0$ for every $i$, then $\max_im_i(t)\to\infty$. For $d=1$: if
    $\|\varphi\|_1\ge1$ and $\mu>0$, then $\E[\lambda(t)]\to\infty$.
\end{enumerate}
\end{proposition}

\begin{proof}
For $\theta>0$ let $\hat\Phi(\theta)=\int_0^\infty e^{-\theta u}\Phi(u)\,\dd u$. By dominated convergence
$\hat\Phi(\theta)\to0$, so we may fix $\theta$ with $c=\max_i\sum_j\hat\Phi_{ij}(\theta)<1$. The Laplace transform
of a convolution is the product of the transforms (Tonelli), so for every locally bounded $f\ge0$,
\begin{equation}\label{eq:hk-weighted}
  e^{-\theta t}(\Phi^{*n}*f)(t)\le\hat\Phi(\theta)^n\sup_{s\le t}e^{-\theta s}f(s)\quad\text{entrywise,}\qquad
  \|\hat\Phi(\theta)^n\|_\infty\le c^n .
\end{equation}

\emph{Finite mean.} Let $n_k(t)=\E[N_{t\wedge t_k}]$ (a vector, with entries at most $k$). For $s\le t_k$ the
events before $s$ are among the first $k$, so by \cref{lem:hk-compensator}(i) and Tonelli,
\[
  n^i_k(t)\le\mu_it+\sum_j\E\int_0^t\!\!\int_{(0,s)}\varphi_{ij}(s-u)\,\dd N^j_{u\wedge t_k}\,\dd s
  =\mu_it+\sum_j\int_0^t\varphi_{ij}(t-s)\,n^j_k(s)\,\dd s ,
\]
using $\int_{(0,t)}\int_u^t\varphi(s-u)\,\dd s\,\dd N_u=\int_{(0,t)}\int_u^t\varphi(t-s)\,\dd s\,\dd N_u=\int_0^t\varphi(t-s)N_s\,\dd s$.
Multiply by $e^{-\theta t}$, bound $te^{-\theta t}\le1/(e\theta)$, and let $a_k(t)=\sup_{s\le t}\max_ie^{-\theta s}n^i_k(s)$:
then $a_k(t)\le\max_i\mu_i/(e\theta)+c\,a_k(t)$, so $\E[N^i_t]=\lim_kn^i_k(t)\le e^{\theta t}\max_i\mu_i/(e\theta(1-c))$.
By \cref{lem:hk-compensator}(ii), $N$ does not explode.

\emph{The mean.} Now the same computation holds with equalities: $M(t)=\E[N_t]$ is finite, nondecreasing and
solves $M(t)=\mu t+(\Phi*M)(t)$. Iterating $n$ times, $M=\sum_{l<n}\Phi^{*l}*(\mu\,\mathrm{id})+\Phi^{*n}*M$, and
the remainder tends to $0$ by \eqref{eq:hk-weighted}; since $(\Phi^{*l}*\mathrm{id})(t)=\int_0^t\int_0^u\Phi^{*l}$,
$M(t)=\int_0^tm(u)\,\dd u$, with $m$ finite. By \cref{lem:hk-compensator}(ii) with
$g(s)=\varphi_{ij}(t-s)\1_{\{s<t\}}$, $\E[\lambda(t)]=\mu+(\Phi*m)(t)$, and $\mu+\Phi*m=m$ term by term. If $\tilde m$ is
another locally bounded solution, $m-\tilde m=\Phi^{*n}*(m-\tilde m)$ for all $n$, and \eqref{eq:hk-weighted}
applied to $|m-\tilde m|$ gives $\tilde m=m$.

(ii) By Tonelli, $\int_0^\infty\Phi^{*n}=G^n$, and $\int_0^t\Phi^{*n}\uparrow G^n$. If $\varrho(G)<1$ the
Neumann series $\sum_nG^n$ converges to $(I-G)^{-1}$, so $m(t)\uparrow\Lambda$ by monotone convergence, and
$\E[N_t]/t=\frac1t\int_0^tm\to\Lambda$.

(iii) If $G^n\to0$, then $G^nw=z^nw\to0$ for an eigenvector $w$ with eigenvalue $|z|=\varrho(G)$, so $\varrho(G)<1$.
Hence if $\varrho(G)\ge1$ some entry $(G^n)_{ij}$ exceeds some $\varepsilon>0$ for infinitely many $n$, and
$\lim_tm_i(t)=\sum_n(G^n\mu)_i\ge\sum_n(G^n)_{ij}\mu_j=\infty$.
\end{proof}

So the process started empty settles at the rate $\Lambda$ when $\varrho(G)<1$, and, if every $\mu_i>0$, its expected
rate grows without bound otherwise. On the whole line, \textcite{hawkes-1971} computed the mean rate $(I-G)^{-1}\mu$ of a stationary
version (his equation~(21)), and \textcite{hawkes-oakes-1974} built the stationary univariate process as a cluster
process, the construction we now give on $[0,T]$.

\begin{proposition}[Branching structure]\label{prop:hk-branching}
Let $\mu$ and $\varphi_{ij}$ be as in \cref{def:hk-hawkes}, and $T>0$. Let \emph{generation $0$} consist of
independent Poisson processes on $[0,T]$ with rates $\mu_i$ (the \emph{immigrants} of type $i$); given generations
$0,\dots,g$, let every event $(s,j)$ of generation $g$ have, independently, type-$i$ \emph{children} forming a
Poisson process on $(s,T]$ with intensity $\varphi_{ij}(\cdot-s)$, for each $i$; these form generation $g+1$.
\begin{enumerate}[label=(\roman*)]
  \item The set of all events is almost surely finite and, forgetting the generations, it has the law of the
    Hawkes process with baseline $\mu$ and kernels $\varphi_{ij}$ on $[0,T]$.
  \item On $[0,\infty)$, an event of type $j$ has a Poisson$(G_{ij})$ number of type-$i$ children. If
    $\varrho(G)<1$, its expected number of type-$i$ descendants, itself included, is $[(I-G)^{-1}]_{ij}$, and its
    \emph{cluster} of descendants is almost surely finite.
\end{enumerate}
\end{proposition}

\begin{proof}
(ii) The number of points of a Poisson process with intensity $\varphi_{ij}(\cdot-s)$ is Poisson with mean
$G_{ij}$. Let $Z_g\in\N^d$ count the descendants of each type in generation $g$ of one type-$j$ event, $Z_0=e_j$.
Given $Z_g$, the children are independent Poisson counts, so $\E[Z_{g+1}\mid Z_g]=GZ_g$ and $\E[Z_g]=G^ge_j$.
By Tonelli, $\E\sum_gZ_g=\sum_gG^ge_j=(I-G)^{-1}e_j<\infty$, so the cluster is finite almost surely.

(i) Conditioning on generation $g$, the expected density of generation $g+1$ is $\Phi*\rho_g$ if $\rho_g$ is that
of generation $g$; with $\rho_0=\mu$, the expected number of events in $[0,T]$ is
$\sum_g\int_0^T\1^\top(\Phi^{*g}*\mu)=\1^\top\E[N_T]<\infty$ by \cref{prop:hk-stationary}, so it is finite almost
surely. Now fix $0<s_1<\dots<s_n\le T$ and types $j_1,\dots,j_n$. A \emph{genealogy} is a map
$\pi:\{1,\dots,n\}\to\{0,\dots,n\}$ with $\pi(k)<k$, where $\pi(k)=0$ means that event $k$ is an immigrant and
$\pi(k)=l\ge1$ that its parent is event $l$. The construction uses independent Poisson processes: one per type
for the immigrants, and one per type and per event for its children. A Poisson process with intensity $h$ on an
interval $I$ has exactly the points $\dd r_1,\dots,\dd r_q$ with probability $\prod_kh(r_k)\,\dd r_k\,e^{-\int_Ih}$
(\cref{lem:hk-compensator}(iii) with a deterministic intensity). Multiplying over all these processes, the
probability that the events are exactly at $\dd s_1,\dots,\dd s_n$, with these types and genealogy $\pi$, is
\[
  \prod_{k=1}^nw_k(\pi(k))\,\dd s_k\ \exp\Big(-T\sum_i\mu_i-\sum_{l=1}^n\sum_i\int_{s_l}^T\varphi_{ij_l}(u-s_l)\,\dd u\Big),
\]
with $w_k(0)=\mu_{j_k}$ and $w_k(l)=\varphi_{j_kj_l}(s_k-s_l)$. The exponent is $-\int_0^T\bar\lambda(u)\,\dd u$ for
the Hawkes intensity \eqref{eq:hk-hawkes} of the configuration. Each $\pi(k)$ ranges over $\{0,\dots,k-1\}$
independently of the others, so the sum over all genealogies factorizes:
$\sum_\pi\prod_kw_k(\pi(k))=\prod_k\sum_{l<k}w_k(l)=\prod_k\lambda^{j_k}(s_k)$. This is \eqref{eq:hk-density}.
The law of a finite point process on $[0,T]$ is the law of $(N_T,t_1,\kappa_1,\dots,t_{N_T},\kappa_{N_T})$, which
these densities determine.
\end{proof}

The proof is the likelihood of \cref{sec:hk-estimation} read backwards. Each immigrant starts a Galton--Watson
tree in continuous time, with Poisson offspring; the fraction of events that are immigrants is the
\emph{exogenous} part of the activity, $1-\|\varphi\|_1$ for $d=1$ (\exref{ex:hk-cluster}).

\section{Simulation by thinning}\label{sec:hk-simulation}

\textcite{lewis-shedler-1979} simulated an inhomogeneous Poisson process with rate $h\le M$ by \emph{thinning}:
draw a Poisson process of rate $M$ and keep a point at $s$ with probability $h(s)/M$. \textcite{ogata-1981}
extended the idea to intensities that depend on the past, updating the bound after each candidate.

\begin{proposition}[Ogata's thinning algorithm]\label{prop:hk-ogata}
Let $(\lambda^i)$ be a conditional intensity such that, for every history $h$ with last event at $s_k$,
$t\mapsto\bar\lambda(t\mid h)$ is nonincreasing on $(s_k,\infty)$, with finite right limits. Start from $s=0$ and
the empty history $h$, and repeat:
\begin{enumerate}[label=\arabic*.]
  \item set $M=\bar\lambda(s+\mid h)$, draw $E\sim\mathrm{Exp}(1)$ and set $s\leftarrow s+E/M$; stop if $s>T$;
  \item draw $U$ uniform on $[0,1]$; if $UM\le\bar\lambda(s\mid h)$, accept $s$ as an event of the type $i$ with
    $\sum_{l<i}\lambda^l(s\mid h)<UM\le\sum_{l\le i}\lambda^l(s\mid h)$, and append $(s,i)$ to $h$.
\end{enumerate}
The accepted events, with fresh variables $E$ and $U$ at every step, form a point process on $[0,T]$ with
conditional intensity $(\lambda^i)$.
\end{proposition}

\begin{proof}
Fix a history $h$ with last event $s_k$, a type $i$ and $t>s_k$, and write $\lambda^l(u)=\lambda^l(u\mid h)$. While no
new event is accepted, the algorithm restarts at each candidate $s$ with fresh variables, so the probability
$F(s)$ that the next acceptance lies in $(s,t]$ and has type $i$ depends only on the current candidate time
$s\in[s_k,t]$. Conditioning on the next candidate $s'=s+E/M$, with $M=\bar\lambda(s+)\ge\bar\lambda$ on $(s,\infty)$:
it is accepted with type $i$ with probability $\lambda^i(s')/M$, rejected with probability $1-\bar\lambda(s')/M$, so
\[
  F(s)=\int_s^te^{-M(s'-s)}\big(\lambda^i(s')+(M-\bar\lambda(s'))F(s')\big)\,\dd s' .
\]
The function $\tilde F(s)=\int_s^t\lambda^i(u)e^{-\int_s^u\bar\lambda}\dd u$ solves the same equation: it satisfies
$\tilde F'=-\lambda^i+\bar\lambda\tilde F$ and $\tilde F(t)=0$, so $\frac{\dd}{\dd s'}\big(e^{-M(s'-s)}\tilde F(s')\big)
=-e^{-M(s'-s)}(\lambda^i+(M-\bar\lambda)\tilde F)(s')$, and integrating from $s$ to $t$ gives the claim. The difference
$D=F-\tilde F$ is bounded and satisfies $|D(s)|\le\bar M\int_s^t|D(s')|\,\dd s'$ with $\bar M=\bar\lambda(s_k+)$; by
Gronwall's lemma (\cref{lem:sde-gronwall}, applied to $r\mapsto|D(t-r)|$), $D=0$. At $s=s_k$ this is
\eqref{eq:hk-hazard}.
\end{proof}

For a Hawkes process with nonincreasing kernels, $\bar\lambda(\cdot\mid h)$ is nonincreasing between events, and
$\bar\lambda(s+)$ includes the jump of an event accepted at $s$. With exponential kernels the state
$(R_{ij})$ of \cref{rem:hk-exp} is updated in $O(d^2)$ operations per candidate; with general kernels each
evaluation of $\lambda$ costs $O(n)$, and a path with $n$ events $O(n^2)$. When the kernel is not monotone the
bound must be chosen differently (\exref{ex:hk-thinning}). The cluster construction of \cref{prop:hk-branching} is another
exact sampler, which simulates whole generations at once; the code uses it for long or many paths.

\section{Likelihood and goodness of fit}\label{sec:hk-estimation}

\begin{proposition}[Maximum likelihood]\label{prop:hk-mle}
Let a $d$-variate Hawkes process be observed on $[0,T]$, with events $(t_k,\kappa_k)_{k\le n}$, $n=N_T$, and let
its intensity $\lambda_\vartheta$ depend on a parameter $\vartheta$.
\begin{enumerate}[label=(\roman*)]
  \item The log-likelihood is
    \begin{equation}\label{eq:hk-loglik}
      \ell_T(\vartheta)=\sum_{i=1}^d\Big(\sum_{k:\,\kappa_k=i}\log\lambda^i_\vartheta(t_k)-\int_0^T\lambda^i_\vartheta(s)\,\dd s\Big).
    \end{equation}
  \item For exponential kernels, with $R_{ij}(t_1)=0$ and
    $R_{ij}(t_k)=e^{-\beta_{ij}(t_k-t_{k-1})}\big(R_{ij}(t_{k-1})+\1_{\{\kappa_{k-1}=j\}}\big)$,
    \[
      \lambda^i(t_k)=\mu_i+\sum_j\alpha_{ij}\beta_{ij}R_{ij}(t_k),\qquad
      \int_0^T\lambda^i=\mu_iT+\sum_j\alpha_{ij}\sum_{k:\,\kappa_k=j}\big(1-e^{-\beta_{ij}(T-t_k)}\big),
    \]
    so that $\ell_T$ costs $O(nd^2)$ operations \citep{ozaki-1979}.
  \item For fixed decays $\beta_{ij}$, $\ell_T$ is a concave function of $(\mu,\alpha)\in[0,\infty)^{d+d^2}$, and
    a sum of $d$ concave functions, one of $(\mu_i,\alpha_{i1},\dots,\alpha_{id})$ for each $i$.
  \item \emph{(Asymptotics; \textcite{clinet-yoshida-2017}, Theorem~4.6.)} Let the process start empty, let the
    parameter $\vartheta=(\mu,\alpha,\beta)$ of exponential kernels range over a bounded open set $\Theta$ on which
    every $\mu_i,\alpha_{ij},\beta_{ij}$ lies between two positive constants, and let the true $\vartheta^*\in\Theta$ satisfy
    $\varrho(G^*)<1$. Then the maximizer $\hat\vartheta_T$ of $\ell_T$ over the closure of $\Theta$ satisfies
    $\sqrt T(\hat\vartheta_T-\vartheta^*)\to\mathcal N(0,\Gamma^{-1})$ in law, together with its moments, where
    $-\frac1T\nabla^2\ell_T(\vartheta^*)\to\Gamma$ in probability ($\Gamma$ is the Fisher information per unit time).
    Their kernels are $c_{ij}e^{-a_{ij}t}$; the statement transfers through the smooth reparametrisation
    $c=\alpha\beta$, $a=\beta$, with $\Gamma$ changed accordingly. Zero entries $\alpha_{ij}=0$ are excluded.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) is the logarithm of \eqref{eq:hk-density}. (ii) $R_{ij}(t_k)=\sum_{l<k,\,\kappa_l=j}e^{-\beta_{ij}(t_k-t_l)}$:
split off $l=k-1$. Integrate \eqref{eq:hk-hawkes} with $\int_{t_l}^T\beta e^{-\beta(u-t_l)}\dd u=1-e^{-\beta(T-t_l)}$.
(iii) With $\beta$ fixed, each $\lambda^i(t_k)$ is a linear function of $(\mu_i,\alpha_{i\cdot})$, the logarithm is
concave and nondecreasing, and the compensator $\int_0^T\lambda^i$ is linear.
\end{proof}

In practice the covariance of $\hat\vartheta_T$ is estimated by $(-\nabla^2\ell_T(\hat\vartheta_T))^{-1}$, the
inverse \emph{observed} information. \textcite{ogata-1978} proved consistency and asymptotic normality of the
maximum likelihood estimator for stationary point processes; the result above is for the exponential Hawkes process
started empty, as observed in practice; it does not cover the toy order book below, whose $G$ has zero entries.
If $N^i_T\ge1$ and the fitted intensities at the events are positive, the fitted compensator matches the count,
$\int_0^T\hat\lambda^i=N^i_T$: \exref{ex:hk-mle-identity} proves it for fixed decays, and the same scaling argument at
$\hat\beta$ gives it for the full fit.

\begin{theorem}[Random time change]\label{thm:hk-time-change}
Let $N$ have conditional intensity $(\lambda^i)$ with $\int_{s_k}^\infty\bar\lambda(u\mid h)\,\dd u=\infty$ for every
history $h$ (for a Hawkes process: $\sum_i\mu_i>0$), and let $\Lambda(t)=\int_0^t\bar\lambda(u)\,\dd u$.
\begin{enumerate}[label=(\roman*)]
  \item The transformed gaps $\tau_k=\Lambda(t_k)-\Lambda(t_{k-1})$, $k\ge1$, are \iid $\mathrm{Exp}(1)$: the points
    $\Lambda(t_1)<\Lambda(t_2)<\cdots$ form a Poisson process of rate $1$.
  \item \emph{(Multivariate version.)} If $\Lambda^i(t)=\int_0^t\lambda^i\to\infty$ almost surely for each $i$, the
    points $\Lambda^i(t)$ at the type-$i$ event times form $d$ independent Poisson processes of rate $1$
    \citep{meyer-1971,brown-nair-1988}. More generally, (i) holds for every simple point process whose compensator
    is continuous and tends to infinity (\textcite{daley-vere-jones-2003}, Theorem~7.4.I).
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Given $H_{k-1}$, the function $L(t)=\int_{t_{k-1}}^t\bar\lambda(u\mid H_{k-1})\,\dd u$ is continuous,
nondecreasing, and tends to infinity. For $x\ge0$ let $\sigma_x=\sup\{t:L(t)\le x\}$, so that $L(\sigma_x)=x$ and
$\{\tau_k>x\}=\{L(t_k)>x\}=\{t_k>\sigma_x\}$. By \eqref{eq:hk-hazard},
$\Prob(\tau_k>x\mid H_{k-1})=e^{-L(\sigma_x)}=e^{-x}$. This does not depend on $H_{k-1}$, of which
$\tau_1,\dots,\tau_{k-1}$ are functions, so $\tau_k$ is $\mathrm{Exp}(1)$ and independent of
$(\tau_1,\dots,\tau_{k-1})$; by induction the $\tau_k$ are \iid
\end{proof}

For the processes of \cref{def:hk-intensity}, (i) is the construction with exponential variables read in reverse.
Its use is as a \emph{test}: compute $\hat\tau_k$ with the fitted intensity and compare them with $\mathrm{Exp}(1)$,
for instance by a QQ plot and the Kolmogorov--Smirnov statistic (for each type separately, by (ii)). As the
parameters are estimated from the same data, the Kolmogorov--Smirnov $p$-values are approximate.

\Cref{fig:hk-estimation} and \cref{tab:hk-mle} replicate both steps on simulated data with
$(\mu,\alpha,\beta)=(0.5,0.6,2)$ and $T=1000$. Maximum likelihood recovers the parameters, the standard errors from
the observed information match the spread of the estimates, and the time-change test accepts the Hawkes fit
and rejects a Poisson fit by a wide margin.

\section{Order flow in a limit order book}\label{sec:hk-lob}

Many exchanges match orders in a \emph{limit order book}: a limit order offers to buy (a \emph{bid}) or sell (an
\emph{ask}) a quantity at a given price and waits in the queue at that price; a market order buys or sells at once
against the best opposite queue; a waiting order may be cancelled. The best bid and ask define the spread and the
mid-price, which moves when a best queue is depleted. Message data list every event with a time stamp; the LOBSTER
files reconstructed from Nasdaq data, for instance, have the columns (time, event type, order id, size, price,
direction).

Two empirical facts make Hawkes processes natural here. Order flow is self- and cross-exciting: trades cluster
(large orders are split into many small ones) and the book refills after a trade. \textcite{large-2007} fitted a
ten-dimensional Hawkes model with exponential kernels to the London Stock Exchange book and measured this
\emph{resilience}; \textcite{bacry-mastromatteo-muzy-2015}, Section~6, review the order book models that followed.
And the estimated branching ratios are close to one: \textcite{hardiman-bercot-bouchaud-2013} find kernels with
integral close to $1$ for mid-price changes of E-mini S\&P futures over 1998--2011, and the survey reviews similar
findings (Section~3). Most events are then triggered by earlier events rather than by exogenous news.

\paragraph{A toy model.} Take $d=6$ types: market buy (MB) and sell (MS), limit order at the bid (LB) and the ask
(LA), cancellation at the bid (CB) and the ask (CA), with exponential kernels of decay $\beta=4\,\mathrm s^{-1}$ and
the branching matrix of \coderef{code/src/ieor222/hawkes.py}, symmetric between the sides: market orders excite
themselves ($0.65$), a market buy brings new limit orders to the ask ($0.30$), limit orders excite themselves
($0.30$) and their own cancellation ($0.60$), a cancellation is replaced ($0.25$) or attracts a market order on its
side ($0.10$), and a market buy triggers some ask cancellations ($0.05$). Then $\varrho(G)=0.755$, and by
\cref{prop:hk-stationary} the rates are $\Lambda=(0.148,0.148,0.284,0.284,0.218,0.218)$ events per second, $1.30$
in total, against $\1^\top\mu=0.34$ from the baselines: three quarters of the activity is endogenous. A price is
added by a queue mechanism: the best queues hold four lots, limit orders add one, market orders and cancellations
remove one, and when a queue is empty the mid-price moves one tick away from it and both queues restart.
\Cref{fig:hk-lob} shows four simulated hours. Fitting the $42$ parameters by the concave likelihood of
\cref{prop:hk-mle}(iii), with the decay known, recovers every entry of $G$ within $0.035$ (the nonzero ones within $2.3$
standard errors), and the time-change test accepts the fit for each type ($p\ge0.66$).

The model is a toy: the intensities do not see the state of the book, which state-dependent extensions add. What
it shows already is the macroscopic effect of endogeneity. On large time scales a Hawkes process is a drift plus
a Brownian fluctuation, amplified by $(I-G)^{-1}$.

\begin{theorem}[Law of large numbers and central limit theorem]\label{thm:hk-fclt}
Let $N$ be a $d$-variate Hawkes process with $\varrho(G)<1$, $\Lambda=(I-G)^{-1}\mu$ and $\Sigma=\mathrm{diag}(\Lambda)$.
Then $\sup_{v\le1}|T^{-1}N_{Tv}-v\Lambda|\to0$ almost surely and in $L^2$ as $T\to\infty$. If moreover
$\int_0^\infty t^{1/2}\varphi_{ij}(t)\,\dd t<\infty$ for all $i,j$, then
$\big(\sqrt T\,(T^{-1}N_{Tv}-v\Lambda)\big)_{v\in[0,1]}$ converges in law, in the Skorokhod topology, to
$(I-G)^{-1}\Sigma^{1/2}W$, where $W$ is a $d$-dimensional Brownian motion
(\textcite{bacry-delattre-hoffmann-muzy-2013}, Theorems~1--2 and Corollary~1).
\end{theorem}

A price $P=c^\top N$, such as the number of upward minus downward ticks, is thus diffusive on large scales:
$(P_{Tv}-Tv\,c^\top\Lambda)/\sqrt T$ converges to a Brownian motion with variance
$c^\top(I-G)^{-1}\Sigma(I-G)^{-\top}c$ per unit time. The factor $(I-G)^{-1}$ explodes as $\varrho(G)\to1$: in
a nearly unstable book, a few exogenous events generate a large volatility (\exref{ex:hk-price-variance}).

\section{Nearly unstable Hawkes processes and rough volatility}\label{sec:hk-scaling}

If $\varrho(G)$ is close to $1$ and the data span a long period, the relevant asymptotics are not those of
\cref{thm:hk-fclt} but those in which the branching ratio tends to $1$ as the horizon grows.
\textcite{jaisson-rosenbaum-2015} take $d=1$, a kernel $a_T\varphi$ with $\|\varphi\|_1=1$ and $a_T\uparrow1$,
and look at the process on $[0,T]$ in units of $T$. We keep the notation of these papers: in this section $a_T$ is
the branching ratio, $\lambda$ (without superscript) a scaling constant and $\alpha$ a tail exponent; $\lambda^T$ is
the intensity.

\begin{remark}[Where the CIR process comes from]\label{rem:hk-cir}
Let $\varphi(t)=\beta e^{-\beta t}$, so $m=\int_0^\infty t\varphi(t)\,\dd t=1/\beta$. By \cref{rem:hk-exp},
$\dd\lambda^T_t=\beta(\mu-\lambda^T_t)\,\dd t+a_T\beta\,\dd N^T_t=\beta(\mu-(1-a_T)\lambda^T_t)\,\dd t+a_T\beta\,\dd M_t$, with
$M=N^T-\int\lambda^T$. The rescaled intensity $C^T_t=(1-a_T)\lambda^T_{Tt}$ therefore satisfies
\[
  \dd C^T_t=T(1-a_T)\beta\,(\mu-C^T_t)\,\dd t+(1-a_T)a_T\beta\,\dd M_{Tt} .
\]
The martingale part has jumps of size $(1-a_T)a_T\beta\to0$ at rate $T\lambda^T_{Tt}=TC^T_t/(1-a_T)$, so its
squared jumps accumulate at rate $T(1-a_T)a_T^2\beta^2C^T_t$. If $T(1-a_T)\to\lambda>0$, the drift tends to
$\frac\lambda m(\mu-C)$ and the squared noise to $\frac\lambda{m^2}C$: a CIR process with volatility $\sqrt\lambda/m$.
The first two moments of $C^T_1$ solve linear differential equations (\exref{ex:hk-cir}(b)) and converge to those
of the limit, whose law at time $1$ is a Gamma law (\exref{ex:hk-cir}(d)); for $\mu=1$, $\lambda=1$, $\beta=1/2$, the variance is $0.0395$ at $T=50$ and $0.0387$ in the
limit.
\end{remark}

For heavy-tailed kernels the limit involves the \emph{Mittag-Leffler function}
$E_{\alpha,\gamma}(z)=\sum_{n\ge0}z^n/\Gamma(\alpha n+\gamma)$, $E_\alpha=E_{\alpha,1}$, through the kernel
$f^{\alpha,\lambda}(t)=\lambda t^{\alpha-1}E_{\alpha,\alpha}(-\lambda t^\alpha)$ and its integral
$F^{\alpha,\lambda}(t)=1-E_\alpha(-\lambda t^\alpha)$. Near $0$, $f^{\alpha,\lambda}(t)\approx\lambda t^{\alpha-1}/\Gamma(\alpha)$,
the kernel of the Riemann--Liouville process of \cref{def:rv-fbm} with $H=\alpha-\frac12$.

\begin{theorem}[Scaling limits of nearly unstable Hawkes processes]\label{thm:hk-scaling-limit}
For $T>0$ let $N^T$ be a Hawkes process with $d=1$, baseline $\mu_T>0$ and kernel $a_T\varphi$, where
$a_T\in(0,1)$, $a_T\to1$, $\varphi\ge0$ and $\|\varphi\|_1=1$; let $\lambda^T$ be its intensity.
\begin{enumerate}[label=(\roman*)]
  \item \emph{(Light tails; \textcite{jaisson-rosenbaum-2015}, Theorem~2.2.)} Let $\mu_T=\mu>0$,
    $m=\int_0^\infty t\varphi(t)\,\dd t<\infty$, $\varphi$ differentiable with $\varphi'$ bounded and integrable,
    and $T(1-a_T)\to\lambda>0$. Suppose that $T\psi^T(Tx)/\|\psi^T\|_1$ is bounded uniformly in $x\ge0$ and $T$,
    where $\psi^T=\sum_{n\ge1}(a_T\varphi)^{*n}$. Then $C^T_t=(1-a_T)\lambda^T_{Tt}$, $t\in[0,1]$, converges in law,
    in the Skorokhod topology, to the solution of
    \[
      \dd X_t=\frac\lambda m(\mu-X_t)\,\dd t+\frac{\sqrt\lambda}m\sqrt{X_t}\,\dd B_t,\qquad X_0=0,
    \]
    and $\frac{1-a_T}TN^T_{Tt}$ converges in law to $\int_0^tX_s\,\dd s$.
  \item \emph{(Heavy tails; \textcite{jaisson-rosenbaum-2016}, Proposition~3.2 and Theorems~3.1--3.2.)} Let
    $\alpha\in(\frac12,1)$ and suppose that $\alpha x^\alpha\int_x^\infty\varphi(s)\,\dd s\to K>0$ as $x\to\infty$.
    Let $\delta=K\Gamma(1-\alpha)/\alpha$ and $\lambda,\mu^*>0$, with $T^\alpha(1-a_T)\to\lambda\delta$ and
    $T^{1-\alpha}\mu_T\to\mu^*/\delta$. Then $X^T_t=\frac{(1-a_T)\delta}{T^\alpha\mu^*}N^T_{Tt}$, $t\in[0,1]$, is tight in
    the Skorokhod topology, and every limit point is of the form $X_t=\int_0^tY_s\,\dd s$, where $Y$ has
    H\"older continuous paths of every order $<\alpha-\frac12$ and solves, for a Brownian motion $B$,
    \begin{equation}\label{eq:hk-rough-cir}
      Y_t=F^{\alpha,\lambda}(t)+\frac1{\sqrt{\mu^*\lambda}}\int_0^tf^{\alpha,\lambda}(t-s)\sqrt{Y_s}\,\dd B_s .
    \end{equation}
\end{enumerate}
\end{theorem}

The limit in (i) is the CIR process of \cref{prop:ir-cir} with $\kappa=\lambda/m$, $\theta=\mu$, $\sigma=\sqrt\lambda/m$,
started at $0$; its law at time $1$ is the Gamma law with shape $2\mu m$ and scale $(1-e^{-\lambda/m})/(2m)$
(\exref{ex:hk-cir}(d)). The limit in (ii) is a \emph{rough} CIR
process: equivalently (\exref{ex:hk-mittag-leffler}),
\[
  Y_t=\frac1{\Gamma(\alpha)}\int_0^t(t-s)^{\alpha-1}\lambda(1-Y_s)\,\dd s
  +\frac1{\Gamma(\alpha)}\int_0^t(t-s)^{\alpha-1}\frac{\lambda}{\sqrt{\mu^*\lambda}}\sqrt{Y_s}\,\dd B_s ,
\]
the variance equation of the rough Heston model with $H=\alpha-\frac12$. In (ii) the rescaled intensity itself is
typically not tight; only its integral converges. Heavy tails mean long memory, and long memory with near
instability produces roughness.

\textcite{el-euch-fukasawa-rosenbaum-2018} turned this into a model for prices. Let $N^{T,+}$ and $N^{T,-}$ count
upward and downward ticks, $P^T=N^{T,+}-N^{T,-}$, and let $(N^{T,+},N^{T,-})$ be a bivariate Hawkes process with
baseline $\mu_T(1,1)$ and kernel matrix $a_T\Phi$,
\begin{equation}\label{eq:hk-efr}
  \Phi=\begin{pmatrix}\varphi_1&\beta\varphi_2\\\varphi_2&\varphi_1+(\beta-1)\varphi_2\end{pmatrix},\qquad
  \beta\ge1,\quad\|\varphi_1\|_1+\beta\|\varphi_2\|_1=1 .
\end{equation}
The rows of $\Phi$ have the same sum, $\varphi_1+\beta\varphi_2$, so upward and downward ticks have the same
expected rate (no statistical arbitrage). Here $\beta$ is an asymmetry parameter, not a decay rate. If $\beta>1$, a
downward tick triggers more downward ticks than an upward tick triggers upward ones
($\varphi_1+(\beta-1)\varphi_2>\varphi_1$), and more upward ticks than an upward tick triggers downward ones
($\beta\varphi_2>\varphi_2$). The matrix $\int_0^\infty\Phi$ has the
eigenvalue $1$, with right eigenvector $(1,1)$ and left eigenvector $(1,\beta)$, and the eigenvalue
$\|\varphi_1\|_1-\|\varphi_2\|_1$, with left eigenvector $(1,-1)$; so $\varrho(a_T\int\Phi)=a_T$.

\begin{theorem}[Rough Heston as a scaling limit]\label{thm:hk-efr}
\emph{(\textcite{el-euch-fukasawa-rosenbaum-2018}, Theorem~3.1.)} In the model \eqref{eq:hk-efr}, let
$\alpha\in(\frac12,1)$ and $C>0$ with $\alpha x^\alpha\int_x^\infty(\varphi_1+\beta\varphi_2)(s)\,\dd s\to C$, and let
$T^\alpha(1-a_T)\to\lambda^*>0$ and $T^{1-\alpha}\mu_T\to\mu>0$. Then
$\sqrt{(1-a_T)/(\mu T^\alpha)}\,P^T_{tT}$, $t\in[0,1]$, converges in the sense of finite-dimensional
distributions to
\[
  P_t=\frac1{1-(\|\varphi_1\|_1-\|\varphi_2\|_1)}\sqrt{\frac2{1+\beta}}\int_0^t\sqrt{Y_s}\,\dd W_s,
\]
where, with $\lambda=\alpha\lambda^*/(C\Gamma(1-\alpha))$, $\theta=1+\beta$ and
$\nu=\sqrt{(1+\beta^2)/(\lambda^*\mu(1+\beta))}$,
\[
  Y_t=\frac1{\Gamma(\alpha)}\int_0^t(t-s)^{\alpha-1}\lambda(\theta-Y_s)\,\dd s
  +\frac1{\Gamma(\alpha)}\int_0^t(t-s)^{\alpha-1}\lambda\nu\sqrt{Y_s}\,\dd B_s ,
\]
and $W,B$ are Brownian motions with $\dd[W,B]_t=\frac{1-\beta}{\sqrt{2(1+\beta^2)}}\,\dd t$.
\end{theorem}

This is the rough Heston model of \cref{thm:rv-rough-heston}, with $V_0=0$, Hurst exponent $H=\alpha-\frac12$, and a
correlation that is negative exactly when $\beta>1$: the leverage effect comes from the asymmetry between buying
and selling pressure. The same authors obtain the classical Heston model in the light-tailed case (their
Theorem~2.1), and they note that estimated tail exponents $\alpha$ are close to $\frac12$, which gives the small
values of $H$ measured in \cref{ch:19-rough-volatility}. The amplification factor $1/(1-(\|\varphi_1\|_1-\|\varphi_2\|_1))$
is the one of \cref{thm:hk-fclt} along the price direction $(1,-1)$.

\Cref{fig:hk-nearly-unstable} replicates both regimes by simulation. With light tails, $C^T_1$ at $T=50$ is
close to the Gamma law of the CIR limit; the remaining gap (mean $0.410$ against $0.393$, a Kolmogorov--Smirnov
$p$-value of $5\cdot10^{-4}$ with $3000$ paths) is the bias of the prelimit, of order $1/T$: its exact mean is $0.406$. With a heavy-tailed kernel
$\varphi(t)=\alpha(1+t)^{-1-\alpha}$, $\alpha=0.6$ (so $K=\alpha$, $\delta=\Gamma(0.4)$), $\lambda=\mu^*=1$ and $T=10^6$
(about $10^6$ events), the rescaled event rate is far rougher than with light tails. Its variogram matches the
one of a process with $H=\alpha-\frac12=0.1$, once the averaging over bins is taken into account: the log-log
slope of the same variogram would suggest $H\approx0.24$, as averaging flattens rough variograms at small lags.
The mean converges too, slowly: $\E[X^T_1]$, computed from the renewal equation of \cref{prop:hk-stationary}, is
$0.462$ at $T=10^3$ (and $0.457\pm0.012$ over $600$ simulated paths) and $0.442$ at $T=10^4$, against
$\E[X_1]=\int_0^1F^{0.6,1}(s)\,\dd s=0.431$ in the limit.
