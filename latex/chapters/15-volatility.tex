% Chapter 15: Implied volatility and the smile.

The Black--Scholes model prices all options on a stock with one volatility; inverting it on market
prices gives a different volatility for each strike and maturity, the \emph{implied volatility
surface}. We start from what is wrong with the model, characterize the surfaces free of static
arbitrage, fit them with local volatility (Dupire's formula) and stochastic volatility (the Heston
model), explain why exotics must be priced in one model consistent with all the vanillas, and price
variance swaps and the VIX from the smile without a model.

\paragraph{Setting.} Prices are at time $0$. The interest rate $r$ and the dividend yield $q$ are
constants (\cref{prop:rn-dividends}), $C(K,T)$ and $P(K,T)$ are the prices of the call and the put
with strike $K>0$ and maturity $T>0$, and $F_T=S_0e^{(r-q)T}$ is the forward price. The prices
\emph{come from a pricing measure} $\Q$ if $S_T\ge0$, $\EQ[S_T]=F_T$,
$C(K,T)=e^{-rT}\EQ[(S_T-K)^+]$ and $P(K,T)=e^{-rT}\EQ[(K-S_T)^+]$, as in
\cref{def:arb-pricing-measure,def:rn-measure}. The Black--Scholes call price with dividend yield $q$
is (\cref{prop:rn-dividends})
\begin{equation}\label{eq:iv-bs}
  C^{\mathrm{BS}}(K,T,\sigma)=S_0e^{-qT}N(d_+)-Ke^{-rT}N(d_-),\qquad
  d_\pm=\frac{\log(F_T/K)}{\sigma\sqrt T}\pm\frac{\sigma\sqrt T}{2}.
\end{equation}
Since $S_0e^{-qT}e^k=Ke^{-rT}$ for the \emph{log-moneyness} $k=\log(K/F_T)$, it depends only on $k$
and the \emph{total volatility} $s=\sigma\sqrt T$: $C^{\mathrm{BS}}=S_0e^{-qT}\,b(k,s)$ with
\begin{equation}\label{eq:iv-b}
  b(k,s)=N\Big(-\frac ks+\frac s2\Big)-e^kN\Big(-\frac ks-\frac s2\Big),\qquad s>0 .
\end{equation}
We call $c=C/(S_0e^{-qT})$ the \emph{normalized} price. With $d_\pm=-k/s\pm s/2$, we have
$d_-^2-d_+^2=2k$, hence $e^kN'(d_-)=N'(d_+)$ (\cref{prop:bs-greeks}), and $\partial_kb=-e^kN(d_-)$.

\section{The limits of the Black--Scholes model}\label{sec:iv-limits}

In the Black--Scholes model, daily log returns are independent and normal. Returns of stocks and
indices share three \emph{stylised facts} that contradict this \citep{cont-stylized}.
\emph{Heavy tails}: the kurtosis $\E[(y-\E y)^4]/\Var(y)^2$ of daily log returns $y$ is well above the normal
value $3$. \emph{Volatility clustering}: returns are nearly uncorrelated, but their absolute values are
positively correlated over weeks and months. \emph{Leverage effect}: returns are negatively correlated
with future volatility. \Cref{fig:iv-stylised} shows the three on a series simulated from the Heston
model of \cref{sec:iv-heston} under the real-world measure. The first two come from a random variance
alone; the third also needs a correlation between returns and variance.

\begin{proposition}[Mixtures of normals]\label{prop:iv-mixture}
Let $m\in\R$, let $Z_1,\dots,Z_n$ be independent $\mathcal N(0,1)$ variables, independent of a
vector $(V_1,\dots,V_n)$ of nonnegative variables with $\E[V_i^2]<\infty$ and $\E[V_i]>0$, and let
$y_i=m+\sqrt{V_i}\,Z_i$.
\begin{enumerate}[label=(\roman*)]
  \item The kurtosis of $y_i$ is $3\,\E[V_i^2]/\E[V_i]^2\ge3$, with equality if and only if $V_i$ is
    almost surely constant.
  \item For $i\ne j$, $\Cov(y_i,y_j)=0$, while $\Cov\big((y_i-m)^2,(y_j-m)^2\big)=\Cov(V_i,V_j)$.
  \item For all $i,j$ and every bounded Borel $f$, $\Cov\big(y_i,f(V_j)\big)=0$.
\end{enumerate}
\end{proposition}

\begin{proof}
By independence, $\E[f(V)h(Z)]=\E[f(V)]\,\E[h(Z)]$ when both factors are integrable. Hence
$\E[(y_i-m)^2]=\E[V_i]$, $\E[(y_i-m)^4]=3\E[V_i^2]$, and $\E[V_i^2]\ge\E[V_i]^2$ with equality if and
only if $\Var(V_i)=0$. For $i\ne j$, $\E[(y_i-m)(y_j-m)]=\E[\sqrt{V_iV_j}]\,\E[Z_iZ_j]=0$ and
$\E[(y_i-m)^2(y_j-m)^2]=\E[V_iV_j]$. Finally $\E[(y_i-m)f(V_j)]=\E[\sqrt{V_i}f(V_j)]\,\E[Z_i]=0$.
\end{proof}

Returns can be uncorrelated without being independent: a persistent variance produces clustering, a
random one fat tails. By (iii), shocks independent of the variances give no leverage effect; in the
Heston model it comes from a correlation $\rho<0$ between the two shocks (\exref{ex:iv-leverage}).

\section{Implied volatility}\label{sec:iv-implied}

\begin{proposition}[Existence, uniqueness and bounds]\label{prop:iv-existence}
Let $S_0,K,T>0$ and $r,q\in\R$.
\begin{enumerate}[label=(\roman*)]
  \item The map $\sigma\mapsto C^{\mathrm{BS}}(K,T,\sigma)$ is continuous and strictly increasing on
    $(0,\infty)$, with derivative $S_0e^{-qT}N'(d_+)\sqrt T>0$, and tends to
    $(S_0e^{-qT}-Ke^{-rT})^+$ as $\sigma\downarrow0$ and to $S_0e^{-qT}$ as $\sigma\to\infty$. Hence
    $C^{\mathrm{BS}}(K,T,\sigma)=C$ has a solution $\sigma>0$ if and only if
    \begin{equation}\label{eq:iv-bounds}
      \big(S_0e^{-qT}-Ke^{-rT}\big)^+<C<S_0e^{-qT},
    \end{equation}
    and the solution is then unique.
  \item If $C=C(K,T)$ comes from a pricing measure $\Q$, then $C$ lies in the closed interval
    of \eqref{eq:iv-bounds}, and in the open interval as soon as $\Q(S_T>K)>0$ and $\Q(S_T<K)>0$.
  \item Call and put prices that satisfy parity, $C-P=S_0e^{-qT}-Ke^{-rT}$, have the same
    solution: $P=C^{\mathrm{BS}}(K,T,\sigma)-S_0e^{-qT}+Ke^{-rT}$ if and only if $C=C^{\mathrm{BS}}(K,T,\sigma)$.
  \item With $c=C/(S_0e^{-qT})$ and $k=\log(K/F_T)$, the solution satisfies
    \begin{equation}\label{eq:iv-upper}
      \sigma\sqrt T\le-2N^{-1}\Big(\frac{1-c}{1+e^k}\Big),\qquad\text{with equality when }k=0 .
    \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Work with $b(k,s)$, $s=\sigma\sqrt T$. Differentiating \eqref{eq:iv-b}, with
$\partial_sd_\pm=k/s^2\pm\frac12$ and $e^kN'(d_-)=N'(d_+)$, gives $\partial_sb=N'(d_+)>0$ (the vega of
\cref{prop:bs-greeks}, with $S_0e^{-qT}$ in place of $S_0$). As $s\downarrow0$, $d_\pm\to+\infty$ if
$k<0$, $-\infty$ if $k>0$, and $0$ if $k=0$, so $b\to(1-e^k)^+$; as $s\to\infty$, $d_\pm\to\pm\infty$
and $b\to1$. Conclude by the intermediate value theorem.

(ii) By Jensen's inequality, $\EQ[(S_T-K)^+]\ge(F_T-K)^+$, and $(S_T-K)^+\le S_T$; multiply by
$e^{-rT}$, with $e^{-rT}F_T=S_0e^{-qT}$. If $F_T\ge K$, the lower gap $\EQ[(K-S_T)^+]$ is positive when
$\Q(S_T<K)>0$; if $F_T<K$, it is $\EQ[(S_T-K)^+]$, positive when $\Q(S_T>K)>0$. The upper gap is
$\EQ[\min(S_T,K)]\ge K\,\Q(S_T>K)$.
(iii) The Black--Scholes prices satisfy parity (\cref{prop:rn-dividends}).

(iv) Fix $s>0$ and let $\beta(k)=1-b(k,s)-(1+e^k)N(-s/2)$. Then $\beta(0)=1-(2N(s/2)-1)-2N(-s/2)=0$
and $\beta'(k)=e^k\big(N(-k/s-s/2)-N(-s/2)\big)$, positive for $k<0$ and negative for $k>0$. So
$b(k,s)\ge1-(1+e^k)N(-s/2)$, with equality only at $k=0$; at the solution $c=b(k,s)$, which
rearranges to \eqref{eq:iv-upper}.
\end{proof}

\begin{definition}[Implied volatility]\label{def:iv-implied}
Let $C$ be the price of the call with strike $K$ and maturity $T$, and suppose that it satisfies
\eqref{eq:iv-bounds}. Its \emph{implied volatility} $\sigma_{\mathrm{imp}}(K,T)$ is the unique
$\sigma>0$ with $C^{\mathrm{BS}}(K,T,\sigma)=C$ (\cref{prop:iv-existence}). The \emph{total implied
variance} is $w(k,T)=\sigma_{\mathrm{imp}}(K,T)^2T$, as a function of $k=\log(K/F_T)$ and $T$. The
map $(K,T)\mapsto\sigma_{\mathrm{imp}}(K,T)$ is the \emph{implied volatility surface}, and
$K\mapsto\sigma_{\mathrm{imp}}(K,T)$ the \emph{smile} at maturity $T$.
\end{definition}

By (ii) and (iii), prices from a pricing measure under which $S_T$ has a positive density have an
implied volatility, the same for calls and puts. In \textcite{black-scholes}'s model the surface is
flat; for equity indices the smile is a \emph{skew}, decreasing in the strike, more steeply at short
maturities (\textcite{gatheral-volatility-surface}, Chapter~1). Out-of-the-money puts insure against the
large falls of \cref{sec:iv-limits}, and are priced accordingly.

\begin{remark}[Computing implied volatility]\label{rem:iv-solver}
At $k=0$, $\sigma_{\mathrm{imp}}=\frac2{\sqrt T}N^{-1}\big(\frac{1+c}2\big)$ by \eqref{eq:iv-upper}, the exact
form of the rule of \textcite{brenner-subrahmanyam} (\cref{prop:bs-atm}): an at-the-money 3-month option
priced $36$ on a spot of $1000$ ($r=q=0$) has $\sigma_{\mathrm{imp}}=18.05\%$ ($18\%$ by the rule). Otherwise the bound \eqref{eq:iv-upper} (\textcite{tehranchi-bounds}, Proposition~4.3) brackets the root,
and Newton's method for $\log b(k,s)=\log c$, safeguarded by bisection, converges for every admissible
price (\coderef{code/src/ieor222/volatility.py}). Since $\delta\sigma\approx\delta C/\partial_\sigma C^{\mathrm{BS}}$,
price errors are amplified where the vega is small: in the wings and at short maturities.
\end{remark}

\begin{remark}[Bachelier versus Black--Scholes]\label{rem:iv-bachelier}
In \textcite{bachelier-speculation}'s model $S_T=F_T+\sigma_{\mathrm N}\sqrt T\,Z$, $Z\sim\mathcal N(0,1)$,
and the call costs $e^{-rT}\big[(F_T-K)N(d)+\sigma_{\mathrm N}\sqrt TN'(d)\big]$, $d=(F_T-K)/(\sigma_{\mathrm N}\sqrt T)$;
every price in \eqref{eq:iv-bounds} has a \emph{normal implied volatility} $\sigma_{\mathrm N}$. Implied
volatility belongs to a price and a model: at the forward $\sigma_{\mathrm N}\approx F_T\sigma_{\mathrm{imp}}$,
and a flat normal smile is a lognormal skew (\exref{ex:iv-bachelier}).
\end{remark}

\section{Static arbitrage in implied volatility}\label{sec:iv-static}

Call prices must be convex in the strike (\cref{prop:arb-strike}) and, by Jensen's inequality,
nondecreasing in the maturity when the strike moves with the forward. For the total implied variance
these conditions become an inequality in $k$ and a monotonicity in $T$. For a function $w$ of $k$, let
\begin{equation}\label{eq:iv-g}
  g(k)=\Big(1-\frac{kw'(k)}{2w(k)}\Big)^2-\frac{w'(k)^2}4\Big(\frac1{w(k)}+\frac14\Big)+\frac{w''(k)}2 .
\end{equation}

\begin{proposition}[Static arbitrage in total implied variance]\label{prop:iv-static-arbitrage}
Let $w:\R\times(0,T^*]\to(0,\infty)$ and $C(K,T)=S_0e^{-qT}\,b\big(k,\sqrt{w(k,T)}\big)$, $k=\log(K/F_T)$:
the prices whose total implied variance is $w$.
\begin{enumerate}[label=(\roman*)]
  \item \emph{(Butterflies.)} Fix $T$, and let $w=w(\cdot,T)$ be twice continuously differentiable.
    Then $C(\cdot,T)$ is twice continuously differentiable on $(0,\infty)$ and, with
    $d_\pm(k)=-k/\sqrt{w(k)}\pm\sqrt{w(k)}/2$,
    \begin{equation}\label{eq:iv-density}
      \partial_{KK}C(K,T)=e^{-rT}\,\frac{N'(d_-(k))}{K\sqrt{w(k)}}\,g(k).
    \end{equation}
    There is a probability measure $\mu$ on $[0,\infty)$ with $\int x\,\mu(\dd x)=F_T$ and
    $C(K,T)=e^{-rT}\int(x-K)^+\mu(\dd x)$ for all $K>0$ if and only if $g\ge0$ on $\R$ and
    $d_+(k)\to-\infty$ as $k\to\infty$.
  \item \emph{(Calendar spreads.)} If the prices come from a pricing measure $\Q$ under which
    $(e^{-(r-q)t}S_t)_{t\le T^*}$ is a martingale, then $T\mapsto w(k,T)$ is nondecreasing for every $k$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Let $B(k,v)=b(k,\sqrt v)$ and $\psi=\partial_vB=N'(d_+)/(2\sqrt v)$. From $d_\pm=-k/\sqrt v\pm\sqrt v/2$,
$\partial_kb=-e^kN(d_-)$ and $e^kN'(d_-)=N'(d_+)$, a direct computation gives
\[
  \partial_{kk}B-\partial_kB=2\psi,\qquad 2\partial_{kv}B-\partial_vB=-\frac{2k}{v}\,\psi,\qquad
  \partial_{vv}B=\Big(\frac{k^2}{2v^2}-\frac18-\frac1{2v}\Big)\psi ;
\]
for the last, $\log\psi=-\frac12d_+^2-\frac12\log v+\mathrm{const}$ and $d_+\partial_vd_+=\frac18-\frac{k^2}{2v^2}$.
Let $\gamma(k)=B(k,w(k))$. As $K^2\partial_{KK}=\partial_{kk}-\partial_k$,
\[
  K^2\partial_{KK}\gamma=\gamma''-\gamma'
  =(\partial_{kk}B-\partial_kB)+(2\partial_{kv}B-\partial_vB)w'+\partial_{vv}B\,w'^2+\partial_vB\,w''
  =2\psi\,g ,
\]
and $S_0e^{-qT}N'(d_+)=Ke^{-rT}N'(d_-)$ gives \eqref{eq:iv-density}, a continuous function of $K$.

If $\mu$ exists, $C(\cdot,T)$ is convex, so $g\ge0$, and $C(K,T)\to0$ as $K\to\infty$ by dominated
convergence. If $d_+(k_n)\ge-M$ along some $k_n\to\infty$, then $\sqrt{w(k_n)}\to\infty$ (otherwise
$d_+(k_n)\le-k_n/A+A/2\to-\infty$ along a subsequence with $\sqrt{w(k_n)}\le A$) and
$d_-(k_n)\le-\sqrt{w(k_n)}/2\to-\infty$, so, with $N(x)\le N'(x)/|x|$ for $x<0$,
\[
  b\big(k_n,\sqrt{w(k_n)}\big)\ge N(-M)-e^{k_n}N(d_-)\ge N(-M)-\frac{N'(d_+)}{|d_-|}\longrightarrow N(-M)>0,
\]
against $C\to0$. Conversely, let $g\ge0$, $d_+\to-\infty$ and $\gamma_T(K)=e^{rT}C(K,T)$. Then
$\gamma_T$ is convex, $(F_T-K)^+<\gamma_T(K)<F_T$ by \cref{prop:iv-existence}(i), and
$\gamma_T(K)\le F_TN(d_+)\to0$ as $K\to\infty$. A convex function with a finite limit at infinity is
nonincreasing, and $\gamma_T'\ge-1$: if $\gamma_T'(K_1)<-1$, then $\gamma_T'<-1$ on $(0,K_1)$ and
$F_T-\gamma_T(K_1)=\int_0^{K_1}(-\gamma_T')>K_1$, against $\gamma_T(K_1)>F_T-K_1$. Let $\mu$ be the
probability measure on $[0,\infty)$ with $\mu((x,\infty))=-\gamma_T'(x)$ for $x>0$ (nonincreasing,
continuous, with limit $0$ at infinity since $|\gamma_T'(x)|\le2\gamma_T(x/2)/x$), and atom
$1+\gamma_T'(0+)$ at $0$. By Tonelli, $\int(y-K)^+\mu(\dd y)=\int_K^\infty\mu((x,\infty))\,\dd x=\gamma_T(K)$
for $K>0$, and $\int y\,\mu(\dd y)=\gamma_T(0+)=F_T$.

(ii) $X_t=e^{-(r-q)t}S_t/S_0=S_t/F_t$ is a martingale. For $T_1<T_2$, conditional Jensen gives
$\EQ[(X_{T_2}-e^k)^+\mid\F_{T_1}]\ge(X_{T_1}-e^k)^+$, so $\EQ[(X_T-e^k)^+]=e^{rT}C(F_Te^k,T)/F_T
=b\big(k,\sqrt{w(k,T)}\big)$ is nondecreasing in $T$, and $b$ is strictly increasing in its second
argument.
\end{proof}

The comparison in (ii) is at fixed log-moneyness. The condition $g\ge0$ is \emph{Durrleman's
condition}, and the conditions of (i) and (ii) are also sufficient: if every slice satisfies (i), $w(k,\cdot)$ is nondecreasing and
$w(k,T)\to0$ as $T\downarrow0$, the prices come from a martingale (\textcite{roper-arbitrage-free},
Theorem~2.9, for $r=q=0$; apply it to $S_t/F_t$ in general). The limit in (i) bounds the wings: $w(k)<2k$
for large $k$, and $w(k)<2|k|$ for $k\ll0$ if $\Q(S_T=0)=0$ (\exref{ex:iv-wings}); in general
$\limsup_{|k|\to\infty}w(k)/|k|\le2$, part of \textcite{lee-moment-formula}'s moment formula.

Quotes exist for finitely many strikes and maturities, and a surface is fitted to them. The
\emph{raw SVI} slice $w(k)=a+b\big(\rho(k-m)+\sqrt{(k-m)^2+\varsigma^2}\big)$ has linear wings, as the
moment formula suggests, but can fit quotes well and still violate (i): in an example of A.~Vogt reported
by \textcite{gatheral-jacquier-2014}, \eqref{eq:iv-density} is negative for $k\in[0.64,1.26]$, so butterflies
centred there have negative prices (\cref{fig:iv-static}). The same authors give the surface
parameterization SSVI, $w(k,T)=\frac{\theta_T}2\big(1+\rho\varphi(\theta_T)k+\sqrt{(\varphi(\theta_T)k+\rho)^2+1-\rho^2}\big)$,
with explicit conditions on $\theta$ and $\varphi$ that exclude both arbitrages (their Theorems~4.1
and~4.2). \Cref{ch:17-arbitrage-free-surfaces} builds arbitrage-free surfaces directly from quotes.

\section{Local volatility and Dupire's formula}\label{sec:iv-dupire}

In the local volatility models of \cref{def:sde-local-vol} the volatility is a function $\sigma(t,S_t)$.
Call prices for all strikes and maturities determine it: \textcite{dupire-smile} observed that the
forward equation of \cref{thm:fk-kolmogorov}, integrated twice in the strike, is an equation for the
call prices, which can be solved for $\sigma$.

\begin{theorem}[Dupire's formula]\label{thm:iv-dupire}
Let $r,q\in\R$, $T^*>0$, and let $\sigma:[0,T^*]\times(0,\infty)\to[0,\infty)$ be continuous, with
$\sigma\le M$ and $|\sigma(t,x)-\sigma(t,x')|\le\ell\,|\log x-\log x'|$ for constants $M,\ell$, and such that
$a=\sigma^2$ has continuous partial derivatives $\partial_xa$, $\partial_{xx}a$. Under $\Q$, let $S$ be the
positive solution of $\dd S_t=(r-q)S_t\,\dd t+\sigma(t,S_t)S_t\,\dd\widetilde W_t$, $S_0>0$, and suppose that
$Y=\log S$ has a transition density $p_Y$ with $(T,\eta)\mapsto p_Y(0,\log S_0;T,\eta)$ in
$C^{1,2}((0,T^*]\times\R)$. Then $C(K,T)=e^{-rT}\EQ[(S_T-K)^+]$ is in $C^{1,2}((0,T^*]\times(0,\infty))$ and
\begin{equation}\label{eq:iv-dupire-pde}
  \partial_TC(K,T)=\frac{\sigma(T,K)^2K^2}{2}\,\partial_{KK}C(K,T)-(r-q)K\,\partial_KC(K,T)-q\,C(K,T).
\end{equation}
In particular, wherever $\partial_{KK}C(K,T)>0$,
\begin{equation}\label{eq:iv-dupire}
  \sigma(T,K)^2=\frac{2\big(\partial_TC+(r-q)K\,\partial_KC+qC\big)}{K^2\,\partial_{KK}C}.
\end{equation}
\end{theorem}

\begin{proof}
\emph{The forward equation for the price.} $Y$ solves
$\dd Y_t=\tilde b(t,Y_t)\,\dd t+\tilde\sigma(t,Y_t)\,\dd\widetilde W_t$ with $\tilde\sigma(t,y)=\sigma(t,e^y)$ and
$\tilde b=r-q-\tilde\sigma^2/2$, which are bounded and Lipschitz in $y$, so it has a unique solution
(\cref{thm:sde-existence}), and $S=e^Y$ by It\^o's formula. As in \cref{ch:09-sde}, $S$ also solves an
equation on $\R$ with Lipschitz coefficients (extend $x\sigma(t,x)$ by $0$ for $x\le0$), so
$\EQ[\sup_{t\le T^*}S_t^2]<\infty$. By \cref{thm:fk-kolmogorov}(ii), $\varrho(T,\eta)=p_Y(0,\log S_0;T,\eta)$
satisfies $\partial_T\varrho=\frac12\partial_{\eta\eta}(\tilde a\varrho)-\partial_\eta(\tilde b\varrho)$, $\tilde a=\tilde\sigma^2$.
The density of $S_T$ is $p(T,x)=\varrho(T,\log x)/x$; with $\nu=r-q$ and $\partial_\eta=x\partial_x$,
$x\,\partial_Tp=\frac12x\partial_x\big(x\partial_x(axp)\big)-x\partial_x(\nu xp)+\frac12x\partial_x(axp)$, and
$x\partial_x(axp)+axp=\partial_x(ax^2p)$, so
\begin{equation}\label{eq:iv-fp-price}
  \partial_Tp=\tfrac12\partial_{xx}\big(ax^2p\big)-\partial_x\big(\nu xp\big)\qquad\text{on }(0,T^*]\times(0,\infty).
\end{equation}

\emph{Integrating twice.} Let $\phi(T,K)=\EQ[(S_T-K)^+]$. For $0<T_1<T_2\le T^*$ and $0<K<R$, integrate
\eqref{eq:iv-fp-price} against $x-K$ over $[T_1,T_2]\times[K,R]$, where all functions are continuous,
exchange the integrals and integrate by parts in $x$:
\[
  \int_K^R(x-K)\big(p(T_2,x)-p(T_1,x)\big)\dd x
  =\int_{T_1}^{T_2}\!\Big(\frac{a(t,K)K^2p(t,K)}2+\nu\int_K^Rxp(t,x)\,\dd x\Big)\dd t+\alpha_R-\beta_RK,
\]
with $\beta_R=\int_{T_1}^{T_2}\big(\frac12\partial_x(ax^2p)-\nu xp\big)(t,R)\,\dd t$ and
$\alpha_R=R\beta_R-\frac12\int_{T_1}^{T_2}(ax^2p)(t,R)\,\dd t$. As $R\to\infty$, by monotone convergence the
left side tends to $\phi(T_2,K)-\phi(T_1,K)$ and the inner integral to $\EQ[S_t\1_{\{S_t>K\}}]\le F_t$.
So $\alpha_R-\beta_RK$ converges for every $K$; with two values of $K$, $\beta_R\to\beta$ and $\alpha_R\to\alpha$:
\begin{equation}\label{eq:iv-dupire-integrated}
  \phi(T_2,K)-\phi(T_1,K)=\int_{T_1}^{T_2}\Big(\frac{a(t,K)K^2p(t,K)}2+\nu\,\EQ\big[S_t\1_{\{S_t>K\}}\big]\Big)\dd t+\alpha-\beta K .
\end{equation}
As $K\to\infty$, the left side and the term in $\nu$ tend to $0$ (dominated convergence, with
$S_t\le\sup_{u\le T^*}S_u$), so $h(K)=\frac12\int_{T_1}^{T_2}a(t,K)K^2p(t,K)\,\dd t=\beta K-\alpha+o(1)$. But
$h\ge0$ and, by Tonelli, $\int_0^\infty h\,\dd K\le\frac{M^2}2\int_{T_1}^{T_2}\EQ[S_t^2]\,\dd t<\infty$; this
forces $\beta=0$, then $\alpha=0$.

\emph{Differentiating.} The integrand of \eqref{eq:iv-dupire-integrated} is continuous in $t$, since $p$ is
and since $S$ has continuous paths and $\Q(S_t=K)=0$ (dominated convergence again). So
$\partial_T\phi=\frac12aK^2p+\nu\,\EQ[S_T\1_{\{S_T>K\}}]$. By \cref{prop:arb-breeden-litzenberger} (with $B=1$),
$\partial_K\phi=-\Q(S_T>K)$ and $\partial_{KK}\phi=p(T,K)$, so $\EQ[S_T\1_{\{S_T>K\}}]=\phi-K\partial_K\phi$ and
$\partial_T\phi=\frac12\sigma(T,K)^2K^2\partial_{KK}\phi+\nu(\phi-K\partial_K\phi)$; with $C=e^{-rT}\phi$ this is
\eqref{eq:iv-dupire-pde}. These derivatives are jointly continuous by the same arguments.
\end{proof}

The numerator of \eqref{eq:iv-dupire} prices a calendar spread per unit of time (adjusted for carry), the
denominator a butterfly: the local variance is positive and finite when the conditions of
\cref{sec:iv-static} hold strictly. In total implied variance,
\begin{equation}\label{eq:iv-dupire-w}
  \sigma\big(T,F_Te^k\big)^2=\frac{\partial_Tw(k,T)}{g(k,T)},
\end{equation}
with $g$ from \eqref{eq:iv-g} on the slice $w(\cdot,T)$ (\exref{ex:iv-dupire-w};
\textcite{gatheral-volatility-surface}, Chapter~1). The hypothesis on the density is that of
\cref{thm:fk-kolmogorov}(ii); for $\sigma$ bounded away from $0$ and with enough bounded H\"older derivatives,
it follows from the adjoint equation of \textcite{friedman-parabolic}, Chapter~1, Section~8.

\begin{remark}[Using the formula]\label{rem:iv-dupire-use}
Read backwards, \eqref{eq:iv-dupire} defines $\sigma$ from call prices. If the model with this $\sigma$
satisfies the hypotheses of the theorem, its prices and the given ones solve \eqref{eq:iv-dupire-pde} with
the limit $(S_0-K)^+$ as $T\downarrow0$, and it reproduces the calls when that solution is unique. As it
differentiates twice, the formula is applied to a smooth arbitrage-free fit, not to raw quotes. Finally,
vanilla prices do not identify the dynamics: if $\dd S_t=(r-q)S_t\,\dd t+\sqrt{v_t}S_t\,\dd\widetilde W_t$,
the local volatility equation with $\sigma(t,x)^2=\EQ[v_t\mid S_t=x]$ has a weak solution with the same
one-dimensional marginals, hence the same call prices (\textcite{gyongy-mimicking}, for bounded,
uniformly non-degenerate coefficients; G.~Brunick and S.~Shreve, \emph{Ann.\ Appl.\ Probab.}, 2013,
under an integrability condition, without uniqueness).
\end{remark}

\section{Stochastic volatility and the Heston model}\label{sec:iv-heston}

Random volatility alone produces a smile. If the variance $v$ is independent of the shocks of the
price, then given its path $\log(S_T/F_T)$ is Gaussian with variance $V=\int_0^Tv_t\,\dd t$ and mean $-V/2$
\citep{hull-white-1987}: $S_T=F_T\exp(\sqrt VZ-V/2)$ with $Z\sim\mathcal N(0,1)$ independent of $V$.

\begin{proposition}[Random variance makes a smile]\label{prop:iv-mixing}
Let $V\ge0$, not almost surely $0$, with $\E[V]<\infty$, be independent of $Z\sim\mathcal N(0,1)$, let the
prices come from the pricing measure under which $S_T=F_T\exp(\sqrt VZ-V/2)$, and let $w(k)$ be the total
implied variance at maturity $T$.
\begin{enumerate}[label=(\roman*)]
  \item (\emph{Mixing.}) $C(K,T)=\EQ\big[S_0e^{-qT}b(k,\sqrt V)\big]$, an average of Black--Scholes prices.
  \item (\emph{Symmetry.}) $w$ is defined on $\R$ and $w(k)=w(-k)$ for every $k$.
  \item (\emph{Smile.}) If $V$ is not almost surely constant, then $w(0)<\EQ[V]$; if moreover $V\le\bar v$,
    then $w(k)>\EQ[V]$ for all $|k|>\sqrt{\bar v+\bar v^2/4}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Given $V=v$, the computation of \cref{thm:bs-formula} with total variance $v$ gives
$\EQ[(S_T-K)^+]=F_Tb(k,\sqrt v)$; apply the independence lemma (\cref{lem:pr-independence}).
(ii) $S_T$ has a positive density on $\{V>0\}$, so $\Q(S_T>K)>0$ and $\Q(S_T<K)>0$, and $w(k)$ exists by
\cref{prop:iv-existence}. By \eqref{eq:iv-b}, $b(k,s)-(1-e^k)=e^kb(-k,s)$ for $s\ge0$: both sides are the
normalized put (put--call symmetry). The prices $\bar c(k)=\EQ[b(k,\sqrt V)]$ satisfy the same identity, so
$b\big(-k,\sqrt{w(k)}\big)=e^{-k}\big(\bar c(k)-1+e^k\big)=\bar c(-k)=b\big(-k,\sqrt{w(-k)}\big)$, and $w(k)=w(-k)$.
(iii) Let $h_k(v)=b(k,\sqrt v)$: continuous on $[0,\infty)$, increasing, and by the proof of
\cref{prop:iv-static-arbitrage}, $h_k''=h_k'\big(\frac{k^2}{2v^2}-\frac18-\frac1{2v}\big)$ on $(0,\infty)$. So
$h_0$ is strictly concave, and Jensen gives $h_0(w(0))=\EQ[h_0(V)]<h_0(\EQ[V])$. If
$|k|>\sqrt{\bar v+\bar v^2/4}$, then $k^2>v+v^2/4$ on $(0,\bar v]$, $h_k$ is strictly convex on $[0,\bar v]$,
and $h_k(w(k))=\EQ[h_k(V)]>h_k(\EQ[V])$.
\end{proof}

Uncorrelated random volatility thus produces a symmetric smile, below the average volatility at the
money and above it in the wings; \textcite{renault-touzi-1996} show that it is U-shaped, with its minimum at
the forward. A skew needs a correlation between the price and its volatility, as in the model of
\textcite{heston-closed-form}, which also keeps prices computable.

\begin{lemma}[A Fourier formula for calls]\label{lem:iv-lewis}
Let $X$ be a random variable with $\E[e^X]=1$, and $\varphi(z)=\E[e^{izX}]$ for complex $z$ with
$-1\le\operatorname{Im}z\le0$. Then for every $k\in\R$,
\[
  \E\big[(e^X-e^k)^+\big]=1-\frac{e^{k/2}}\pi\int_0^\infty
  \operatorname{Re}\Big[e^{-iuk}\varphi\big(u-\tfrac i2\big)\Big]\frac{\dd u}{u^2+\frac14}.
\]
\end{lemma}

\begin{proof}
Write $(e^X-e^k)^+=e^X-\min(e^X,e^k)$ and $\min(e^x,e^k)=e^{k/2}e^{x/2}\psi(x)$ with $\psi(x)=e^{-|x-k|/2}$.
Then $\int e^{iux}\psi(x)\,\dd x=e^{iuk}/(u^2+\frac14)$ is integrable, so by Fourier inversion
(\textcite{durrett}, Section~3.3, for the Laplace density $\psi/4$),
$\psi(x)=\frac1{2\pi}\int e^{-iux}e^{iuk}(u^2+\frac14)^{-1}\dd u$ for every $x$. As
$\E[e^{X/2}]\le\E[e^X]^{1/2}=1$, Fubini's theorem gives
\[
  \E\big[\min(e^X,e^k)\big]=\frac{e^{k/2}}{2\pi}\int_\R e^{iuk}\,\E\big[e^{(1/2-iu)X}\big]\frac{\dd u}{u^2+\frac14},
\]
with $\E[e^{(1/2-iu)X}]=\overline{\varphi(u-\frac i2)}$. The integrand at $-u$ is the conjugate of that at
$u$, so the integral is twice the integral over $(0,\infty)$ of the real part, which is the claim.
\end{proof}

\begin{proposition}[The Heston model]\label{prop:iv-heston}
Let $\kappa,\theta,\xi,v_0>0$ and $\rho\in[-1,1]$. Under $\Q$, let $\widetilde W$ and $B$ be Brownian
motions with $[\widetilde W,B]_t=\rho t$, let $v$ be the square-root (CIR) process
$\dd v_t=\kappa(\theta-v_t)\,\dd t+\xi\sqrt{v_t}\,\dd B_t$, which is nonnegative, and positive if
$2\kappa\theta\ge\xi^2$ (\cref{prop:ir-cir}), and let
\[
  S_t=S_0\exp\Big(\int_0^t\sqrt{v_u}\,\dd\widetilde W_u+\int_0^t\Big(r-q-\frac{v_u}2\Big)\dd u\Big),
  \qquad\dd S_t=(r-q)S_t\,\dd t+\sqrt{v_t}\,S_t\,\dd\widetilde W_t .
\]
\begin{enumerate}[label=(\roman*)]
  \item $\EQ[v_t]=\theta+(v_0-\theta)e^{-\kappa t}$, so that
    $\frac1T\EQ\big[\int_0^Tv_t\,\dd t\big]=\theta+(v_0-\theta)\frac{1-e^{-\kappa T}}{\kappa T}$.
  \item $(e^{-(r-q)t}S_t)$ is a martingale, and for real $u$, $\EQ[e^{iuX_T}]=\exp\big(A(u,T)+D(u,T)v_0\big)$
    for $X_T=\log(S_T/F_T)$, where, with $\beta=\kappa-\rho\xi iu$, $d=\sqrt{\beta^2+\xi^2(iu+u^2)}$ (principal
    root) and $\gamma=(\beta-d)/(\beta+d)$,
    \begin{equation}\label{eq:iv-heston-cf}
      D=\frac{\beta-d}{\xi^2}\,\frac{1-e^{-dT}}{1-\gamma e^{-dT}},\qquad
      A=\frac{\kappa\theta}{\xi^2}\Big((\beta-d)T-2\log\frac{1-\gamma e^{-dT}}{1-\gamma}\Big).
    \end{equation}
  \item If $\rho=0$, the smile is symmetric in $k$, and below
    $\big(\frac1T\EQ\int_0^Tv_t\,\dd t\big)^{1/2}$ at $k=0$, at every maturity.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) As $v\ge0$, $v_t\le v_0+\kappa\theta t+\xi M_t$ with $M=\int_0^\cdot\sqrt v\,\dd B$ a local martingale;
localizing (\cref{rem:ito-local}) and applying Fatou's lemma, $\EQ[v_t]\le v_0+\kappa\theta t$. So $M$ is a
martingale (\cref{thm:ito-integral}), and $m(t)=\EQ[v_t]$ solves $m(t)=v_0+\int_0^t\kappa(\theta-m(u))\,\dd u$,
as for the Ornstein--Uhlenbeck mean in \cref{ch:09-sde}. (ii) The martingale property is Proposition~2.5
of \textcite{andersen-piterbarg-2007}. The characteristic function is equation~(17) of
\textcite{heston-closed-form}, without the drift $(r-q)T$, in the equivalent form of
\textcite{albrecher-heston-trap}, which, unlike Heston's, does not cross the branch cut of the complex
logarithm as $u$ or $T$ grows. (iii) If $\rho=0$, $(\widetilde W,B)$ is a two-dimensional Brownian motion
(\cref{thm:itf-levy}). By pathwise uniqueness (\cref{prop:ir-cir}(i)) and the Yamada--Watanabe theorem
(\textcite{karatzas-shreve}, Section~5.3.D), $v$ is a functional of $B$, hence independent of $\widetilde W$,
and given $v$, $\log(S_T/F_T)$ is $\mathcal N(-V/2,V)$ (\textcite{romano-touzi-1997}, who make
\textcite{hull-white-1987}'s argument rigorous); $V$ is not constant. Apply \cref{prop:iv-mixing}.
\end{proof}

Behind (ii) is the affine structure of \cref{thm:ir-affine}: $A$ and $D$ solve Riccati equations
(\exref{ex:iv-heston-riccati}). Prices use \cref{lem:iv-lewis} (a formula of A.~Lewis, 2001). As
$\EQ[e^{pX_T}]\le1$ for $p\in[0,1]$, $\varphi$ is analytic on $-1<\operatorname{Im}z<0$ and continuous on the
closure; at $z=u-\frac i2$, \eqref{eq:iv-heston-cf} with $iz+z^2=u^2+\frac14$, $\beta=\kappa-\frac{\rho\xi}2-i\rho\xi u$
is this continuation if its root and logarithm cross no branch cut (\textcite{albrecher-heston-trap},
Theorem~3; R.~Lord and C.~Kahl, \emph{Math.\ Finance}, 2010). We checked this for every parameter set
used here against the Riccati equations solved numerically (to $10^{-11}$). $v_0$ and $\theta$ set the short and long
levels, $\rho$ tilts the smile, $\xi$ curves it. The variance is not traded, so risk-neutral measures are
many (\exref{ex:rn-incomplete}); \emph{calibration} to quoted vanillas selects one.

\section{One model for all prices}\label{sec:iv-consistency}

Pricing an exotic by Black--Scholes ``at the right volatility'' fails: the price depends on the shape
of the surface, not only on its level. The digital call is the simplest example.

\begin{proposition}[A digital under the smile]\label{prop:iv-digital}
Let the prices $C(\cdot,T)$ come from a pricing measure $\Q$ with $\Q(S_T=K)=0$, and let
$\sigma_{\mathrm{imp}}(\cdot,T)$ be differentiable at $K$. The digital call paying $\1_{\{S_T>K\}}$ costs
\[
  e^{-rT}\Q(S_T>K)=-\partial_KC(K,T)=e^{-rT}N(d_-)-\mathcal V\,\partial_K\sigma_{\mathrm{imp}}(K,T),
\]
where $d_\pm$ and $\mathcal V=S_0e^{-qT}N'(d_+)\sqrt T$ are those of \eqref{eq:iv-bs} at $\sigma=\sigma_{\mathrm{imp}}(K,T)$.
\end{proposition}

\begin{proof}
By \cref{prop:arb-breeden-litzenberger}(i), the one-sided derivatives of $C(\cdot,T)$ at $K$ are
$-e^{-rT}\Q(S_T>K)$ and $-e^{-rT}\Q(S_T\ge K)$, which coincide. Differentiate
$C(K,T)=C^{\mathrm{BS}}(K,T,\sigma_{\mathrm{imp}}(K,T))$, with $\partial_KC^{\mathrm{BS}}=-e^{-rT}N(d_-)$ and
$\partial_\sigma C^{\mathrm{BS}}=\mathcal V$ (\cref{prop:bs-greeks}, with $S_0e^{-qT}$ for $S_0$).
\end{proof}

In the Heston market of \cref{fig:iv-heston} ($\rho=-0.7$, $T=1$), at $K=F_T=102.02$ the implied
volatility is $18.05\%$, falling by $2.26$ points per $10$ of strike, and $\mathcal V=39.7$: the
Black--Scholes digital at $18.05\%$ costs $0.455$, the correct price $0.545$. Quoting digitals that way
can be arbitraged: $1/h$ call spreads between $K$ and $K+h$ pay at most $\1_{\{S_T>K\}}$
(\exref{ex:arb-digital}) and cost $0.534$ for $h=1$; buying the digital and selling them collects $0.079$.

Path-dependent claims depend on more than the vanillas. The local volatility of the Heston surface
(\cref{rem:iv-dupire-use}) has the same vanilla prices, as \cref{fig:iv-local-vol} confirms, but an
up-and-out call with barrier $120$ costs $3.32$ in the Heston model and $2.78$ with local volatility (Monte Carlo, $\pm0.03$):
a barrier option depends on the law of the whole path. Hedges differ too, since a smile-model delta
includes the move of the implied volatility with the spot, which \textcite{hagan-smile-risk} argue local
volatility gets wrong.

The lesson is a consistency principle. (a) Exotics and hedges must be priced in one model, under one
pricing measure calibrated to all the vanillas; prices with a volatility per product are not
expectations under one measure, and \cref{lem:arb-pricing-measure} no longer protects them. (b) The
calibration surface must be free of static arbitrage, or \eqref{eq:iv-dupire} fails. (c) Models fitting the same vanillas still disagree on
exotics; their spread, or model-free bounds such as those of \textcite{brown-hobson-rogers} for barrier
options, measure this model risk.
\Cref{ch:17-arbitrage-free-surfaces} builds arbitrage-free surfaces from bid and ask quotes, the first
step of this programme.

\section{Variance swaps and the VIX}\label{sec:iv-variance}

A \emph{variance swap} pays at $T$ the realized variance $\frac1T\sum_i(\log S_{t_i}/S_{t_{i-1}})^2$ over a
grid of $[0,T]$, usually daily, minus a strike $K_{\mathrm{var}}$. If $\dd S_t=S_t(\alpha_t\,\dd t+\sigma_t\,\dd W_t)$,
it converges in probability to $\frac1T[\log S,\log S]_T=\frac1T\int_0^T\sigma_t^2\,\dd t$ as the grid is
refined (\cref{def:ito-covariation,prop:itf-decomposition}). We price this limit, without a model for
$\sigma$ \citep{carr-madan,demeterfi-variance-swaps}.

\begin{proposition}[Variance swaps]\label{prop:iv-variance-swap}
Under $\Q$, let $S_t=S_0\exp\big(\int_0^t\sigma_u\,\dd\widetilde W_u+\int_0^t(r-q-\frac12\sigma_u^2)\dd u\big)$,
with $\sigma$ progressively measurable and bounded, so that $(e^{-(r-q)t}S_t)$ is a martingale (Novikov's
condition, as in \cref{thm:rn-girsanov}) and $\EQ[S_T]=F_T$, and let the option prices come from $\Q$.
\begin{enumerate}[label=(\roman*)]
  \item Along every path, $\int_0^T\sigma_t^2\,\dd t=2\int_0^T\sigma_t\,\dd\widetilde W_t-2\log(S_T/F_T)$.
  \item The fair strike, which makes the swap worth $0$, is
    \begin{equation}\label{eq:iv-kvar}
      K_{\mathrm{var}}=\frac{\EQ\int_0^T\sigma_t^2\,\dd t}{T}=-\frac2T\EQ\Big[\log\frac{S_T}{F_T}\Big]
      =\frac{2e^{rT}}T\Big(\int_0^{F_T}\!\frac{P(K,T)}{K^2}\,\dd K+\int_{F_T}^\infty\!\frac{C(K,T)}{K^2}\,\dd K\Big).
    \end{equation}
  \item The swap is replicated at zero cost by $\frac{2\,\dd K}{TK^2}$ puts at strikes $K<F_T$ and calls at
    $K>F_T$, $\frac2{TF_T}$ short forwards with delivery price $F_T$ (together worth
    $e^{-rT}K_{\mathrm{var}}$), a loan of $e^{-rT}K_{\mathrm{var}}$, and the self-financing strategy started
    from $0$ that holds $\Delta_t=2e^{-r(T-t)}/(TS_t)$ shares, dividends reinvested in the money market.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) $\log S_T=\log F_T+\int_0^T\sigma\,\dd\widetilde W-\frac12\int_0^T\sigma^2\,\dd t$. (ii) The stochastic
integral has mean $0$ (\cref{thm:ito-integral}), and $\log(S_T/F_T)$ is integrable by (i). Apply \cref{cor:arb-strip} with $B=e^{-rT}$, $f(x)=-\log(x/F_T)$ and $\kappa=F_T$: $f(\kappa)=0$,
$f''(K)=K^{-2}$, and the forward term vanishes since the claim paying $S_T$ costs $e^{-rT}F_T$.
Condition \eqref{eq:arb-strip-integrable} holds: by Tonelli and \cref{thm:arb-carr-madan}, the sum of
the two integrals is $e^{-rT}\EQ[f(S_T)+(S_T-F_T)/F_T]=e^{-rT}\EQ[f(S_T)]<\infty$.
(iii) With dividends reinvested, $\dd(e^{-rt}X_t)=\Delta_te^{-rt}S_t\sigma_t\,\dd\widetilde W_t$
(the computation of \cref{prop:rn-dividends}(i), with $\sigma_t$ for $\sigma$), here $\frac2Te^{-rT}\sigma_t\,\dd\widetilde W_t$, so
$X_T=\frac2T\int_0^T\sigma\,\dd\widetilde W$. By \cref{thm:arb-carr-madan} the options pay
$\frac2T\big(f(S_T)+(S_T-F_T)/F_T\big)$ and the forwards $-\frac2T(S_T-F_T)/F_T$; by (i) the total is
$\frac1T\int_0^T\sigma^2\,\dd t-K_{\mathrm{var}}$.
\end{proof}

The strike averages the smile with weights $1/K^2$; a skew lifts it above the at-the-money implied
variance (\cref{fig:iv-variance-swap}). If $S$ jumps, (i) fails: a jump $J$ of $\log S$ adds $J^2/T$ to the
realized variance but $2(e^J-1-J)/T=(J^2+J^3/3+\dots)/T$ to the hedge.

\begin{definition}[The VIX]\label{def:iv-vix}
For a maturity $T$, let $K_1<\dots<K_n$ be the listed strikes, $F$ the forward price, $R$ the interest
rate, $K_0$ the largest strike $\le F$, and $Q(K_i)$ the mid-quote of the put if $K_i<K_0$, of the call if
$K_i>K_0$, and their average at $K_0$; let $\Delta K_i=\frac12(K_{i+1}-K_{i-1})$ (one-sided at the ends).
The CBOE variance estimate is
\begin{equation}\label{eq:iv-vix}
  \sigma^2_T=\frac2T\sum_i\frac{\Delta K_i}{K_i^2}\,e^{RT}Q(K_i)-\frac1T\Big(\frac F{K_0}-1\Big)^2 .
\end{equation}
The \emph{VIX} is $100\sqrt{\frac{365}{30}V_{30}}$, where $V_{30}$ is the linear interpolation, in the time
to maturity measured in minutes, of the total variances $T_1\sigma^2_{T_1}$ and $T_2\sigma^2_{T_2}$ of S\&P~500
options with maturities $T_1<T_2$ around $30$ days, at $30$ days \citep{cboe-vix}; further rules select
the strikes and maturities.
\end{definition}

The VIX is a discretized $30$-day variance-swap strike. Indeed, \cref{cor:arb-strip} with $\kappa=K_0$
gives, with $x=F/K_0-1$,
\[
  -\frac2T\EQ\Big[\log\frac{S_T}{F}\Big]=\frac{2e^{RT}}T\Big(\int_0^{K_0}\frac{P(K)}{K^2}\,\dd K
  +\int_{K_0}^\infty\frac{C(K)}{K^2}\,\dd K\Big)-\frac2T\big(x-\log(1+x)\big),
\]
and $x-\log(1+x)=\frac{x^2}2+O(x^3)$. The sum in \eqref{eq:iv-vix} is the trapezoidal rule for the
integral, averaging put and call at $K_0$, where the integrand jumps. Up to terms of order $x$, its
derivative jumps there by $-e^{-RT}/K_0^2$ (put and call slopes differ by $e^{-RT}$), so the error formula
of the trapezoidal rule gives an upward bias of about $h^2/(6TK_0^2)$ for strikes spaced by $h$
(\cref{fig:iv-variance-swap}). With jumps, $\mathrm{VIX}^2$ prices the log contract.
